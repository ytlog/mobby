cmake_minimum_required(VERSION 3.22)
project(mobby_llama LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(ANDROID)
    option(MOBBY_REQUIRE_VULKAN "Fail the build when Vulkan shaders cannot be compiled" OFF)
    find_program(MOBBY_GLSLC NAMES glslc HINTS /usr/local/bin /opt/homebrew/bin NO_CMAKE_FIND_ROOT_PATH)
    find_path(MOBBY_VULKAN_HEADERS vulkan/vulkan.hpp HINTS /usr/local/include /opt/homebrew/include NO_CMAKE_FIND_ROOT_PATH)
    set(_mobby_vulkan_enabled OFF)
    if(MOBBY_GLSLC AND MOBBY_VULKAN_HEADERS)
        execute_process(COMMAND "${MOBBY_GLSLC}" -fshader-stage=compute --target-env=vulkan1.3
            -o "${CMAKE_BINARY_DIR}/mobby-vulkan-check.spv"
            "${CMAKE_CURRENT_LIST_DIR}/../../../../../third_party/llama.cpp/ggml/src/ggml-vulkan/vulkan-shaders/flash_attn_decode_phase_2.comp"
            RESULT_VARIABLE _mobby_glslc_result ERROR_VARIABLE _mobby_glslc_error)
        if(_mobby_glslc_result EQUAL 0)
            set(_mobby_vulkan_enabled ON)
            set(Vulkan_GLSLC_EXECUTABLE "${MOBBY_GLSLC}" CACHE FILEPATH "Vulkan shader compiler" FORCE)
            set(SPIRV-Headers_DIR "${CMAKE_CURRENT_LIST_DIR}/cmake" CACHE PATH "Android NDK SPIR-V headers" FORCE)
        endif()
    endif()
    if(MOBBY_REQUIRE_VULKAN AND NOT _mobby_vulkan_enabled)
        message(FATAL_ERROR "Vulkan requires recent host glslc (shaderc) and Vulkan-Headers with vulkan.hpp")
    endif()
    if(NOT _mobby_vulkan_enabled)
        message(WARNING "Vulkan shaders unavailable; building CPU-only llama.cpp")
    endif()
    set(GGML_VULKAN ${_mobby_vulkan_enabled} CACHE BOOL "Build Vulkan backend" FORCE)
endif()
set(LLAMA_SRC "${CMAKE_CURRENT_LIST_DIR}/../../../../../third_party/llama.cpp")
if(NOT EXISTS "${LLAMA_SRC}/include/llama.h")
    message(FATAL_ERROR "Initialize third_party/llama.cpp submodule before building")
endif()
set(LLAMA_BUILD_COMMON ON CACHE BOOL "Build llama.cpp chat template and tool parser" FORCE)
add_subdirectory("${LLAMA_SRC}" "${CMAKE_BINARY_DIR}/llama")
if(ANDROID AND _mobby_vulkan_enabled)
    target_sources(ggml-vulkan PRIVATE vulkan_compat.cpp)
    target_include_directories(ggml-vulkan BEFORE PRIVATE "${MOBBY_VULKAN_HEADERS}")
    target_link_libraries(ggml-vulkan PRIVATE SPIRV-Headers::SPIRV-Headers dl)
endif()
add_library(mobby_llama SHARED llama_jni.cpp)
target_compile_definitions(mobby_llama PRIVATE MOBBY_VULKAN_PACKAGED=$<BOOL:${_mobby_vulkan_enabled}>)
target_include_directories(mobby_llama PRIVATE "${LLAMA_SRC}/include" "${LLAMA_SRC}/ggml/include" "${LLAMA_SRC}/common")
target_link_libraries(mobby_llama PRIVATE llama-common llama log)
if(ANDROID AND _mobby_vulkan_enabled)
    target_include_directories(mobby_llama BEFORE PRIVATE "${MOBBY_VULKAN_HEADERS}")
    target_link_libraries(mobby_llama PRIVATE vulkan)
endif()
