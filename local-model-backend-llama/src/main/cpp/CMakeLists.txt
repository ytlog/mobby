cmake_minimum_required(VERSION 3.22)
project(mobby_llama LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(LLAMA_SRC "${CMAKE_CURRENT_LIST_DIR}/../../../../third_party/llama.cpp")
if(NOT EXISTS "${LLAMA_SRC}/include/llama.h")
    message(FATAL_ERROR "Initialize third_party/llama.cpp submodule before building")
endif()
set(LLAMA_BUILD_COMMON ON CACHE BOOL "Build llama.cpp chat template and tool parser" FORCE)
add_subdirectory("${LLAMA_SRC}" "${CMAKE_BINARY_DIR}/llama")
add_library(mobby_llama SHARED llama_jni.cpp)
target_include_directories(mobby_llama PRIVATE "${LLAMA_SRC}/include" "${LLAMA_SRC}/ggml/include" "${LLAMA_SRC}/common")
target_link_libraries(mobby_llama PRIVATE llama-common llama log)
